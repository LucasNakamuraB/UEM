pub fn get_from_pos_dir(
  grid: Grid,
  pos: Vector,
  direction: Direction,
  tile_size: Int,
) -> Int {
  let grid_x = pos.x / tile_size
  let grid_y = pos.y / tile_size
  case direction {
    Up -> grid |> g_get(grid_x, { grid_y - 1 })
    Down -> grid |> g_get(grid_x, grid_y)
    Left -> grid |> g_get({ grid_x - 1 }, grid_y)
    Right -> grid |> g_get(grid_x, grid_y)
  }
}
